    #video { width: 320px; height: 240px; background: #000; }
    #captureBtn { margin-top: 10px; padding: 10px 20px; font-size: 16px; }
    #status { margin-top: 10px; font-weight: bold; }
  </style>

<body>
  <h1>Vivo Y400 Front Cam Capture</h1>
  <video id="video" autoplay playsinline></video><br/>
  <button id="captureBtn">Capture & Send to Telegram</button>
  <div id="status"></div>

  <script>
    const video = document.getElementById('video');
    const captureBtn = document.getElementById('captureBtn');
    const status = document.getElementById('status');

    // Get camera stream
    navigator.mediaDevices.getUserMedia({ video: true })
      .then(stream => { video.srcObject = stream; })
      .catch(err => { console.error(err); status.textContent = 'Camera access denied'; });

    captureBtn.onclick = async () => {
      status.textContent = 'Capturing...';

      // Capture frame
      const canvas = document.createElement('canvas');
      canvas.width = video.videoWidth;
      canvas.height = video.videoHeight;
      const ctx = canvas.getContext('2d');
      ctx.drawImage(video, 0, 0, canvas.width, canvas.height);

      // Convert to JPEG Base64
      const dataURL = canvas.toDataURL('image/jpeg');

      try {
        const response = await fetch('/capture', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ imageData: dataURL })
        });

        const result = await response.json();
        if (result.success) {
          status.textContent = '✅ Photo sent!';
        } else {
          status.textContent = '❌ ' + result.error;
        }
      } catch (err) {
        console.error(err);
        status.textContent = '❌ Network error';
      }
    };
  </script>
</body>
